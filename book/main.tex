% \bookchapter{chapterNN}: read the chapter from its folder, with its own figure
% and input paths, its macros kept local, and its bibliography at the end.
\makeatletter
\newcommand{\bookchapter}[1]{%
  \graphicspath{{../#1/figures/}}%
  \def\input@path{{../#1/}}%
  \begingroup
  \input{../#1/#1}%
  \printbibliography[heading=subbibintoc,title={References}]%
  \endgroup}
